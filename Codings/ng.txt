import React, {useState} from 'react';
const App = () => { 
  const [name, setName] = useState("");
  return (
    <div>
      <input value = {name} onChange = {(e) => setName(e.target.value)} placeholder = "Enter your name" />
      <p> Hello, {name || "User"}! </p>
      <button onClick = {() => alert(`Hello, ${name}`)}>Greet Me</button>
    </div>
  );
};

export default App;
