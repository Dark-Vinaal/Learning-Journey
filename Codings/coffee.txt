import React, { useState } from "react";

const App = () => {
    const menu = [
        { name: "Espresso", price: 3 },
        { name: "Latte", price: 4 },
        { name: "Cappuccino", price: 4.5 },
        { name: "Americano", price: 3.5 },
        { name: "Mocha", price: 5 }
    ];

    const [order, setOrder] = useState([]);

    return (
        <div>
            <h1>Coffee Shop</h1>

            <h2>Menu</h2>
            {menu.map(coffee => (
                <button onClick={() => setOrder([...order, coffee])}>
                    {coffee.name} - ${coffee.price}
                </button>
            ))}

            <h2>Your Order</h2>
            {order.length ? (
                <>
                    {order.map(coffee => (
                        <p>{coffee.name} - ${coffee.price}</p>
                    ))}
                    <p>Total: ${order.reduce((sum, coffee) => sum + coffee.price, 0)}</p>
                    <button onClick={() => setOrder([])}>Clear Order</button>
                </>
            ) : (
                <p>Your order is empty!</p>
            )}
        </div>
    );
};

export default App;
