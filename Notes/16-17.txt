package main
import "fmt"
func fcfs(bt []int) {
	wt, tat := 0, 0
	fmt.Println("\n--- FCFS Scheduling ---")
	for i, b := range bt {
		tat = wt + b
		fmt.Printf("P%d -> WT: %d TAT: %d\n", i+1, wt, tat)
		wt = tat
	}
}

func roundRobin(bt []int, q int) {
	rem := append([]int{}, bt...)
	time := 0
	fmt.Println("\n--- Round Robin Scheduling ---")
	for {
		done := true
		for i := range rem {
			if rem[i] > 0 {
				done = false
				x := q
				if rem[i] < q {
					x = rem[i]
				}
				time += x
				rem[i] -= x
				if rem[i] == 0 {
					fmt.Printf("P%d completed at %d\n", i+1, time)
				}
			}
		}
		if done {
			break
		}
	}
}

func main() {
	var n, q int
	fmt.Print("Enter number of processes: ")
	fmt.Scan(&n)

	bt := make([]int, n)
	for i := range bt {
		fmt.Print("Enter burst time for P", i+1, ": ")
		fmt.Scan(&bt[i])
	}

	fmt.Print("Enter time quantum: ")
	fmt.Scan(&q)

	fcfs(bt)
	roundRobin(bt, q)
}

package main
import (
	"fmt"
	"sync"
)
type Task struct {
	id   int
	text string
}
type Result struct {
	child int
	task  int
}
func worker(id int, input <-chan Task, output chan<- Result, wg *sync.WaitGroup) {
	defer wg.Done()

	for task := range input {
		fmt.Printf("Child %d received: %s\n", id, task.text)
		output <- Result{id, task.id}
	}

	fmt.Printf("Child %d finished.\n", id)
}
func main() {
	tasks := make(chan Task)
	results := make(chan Result)
	var wg sync.WaitGroup
	for i := 1; i <= 3; i++ {
		wg.Add(1)
		go worker(i, tasks, results, &wg)
	}
	go func() {
		wg.Wait()
		close(results)
	}()
	go func() {
		for i := 1; i <= 6; i++ {
			tasks <- Task{i, fmt.Sprintf("Task %d", i)}
		}
		close(tasks)
	}()
	for result := range results {
		fmt.Printf("Parent received Task %d from Child %d\n",
			result.task, result.child)
	}
	fmt.Println("All child processes completed.")
}
